        async function sendScore() {
            const id = document.getElementById('nickname').value;
            const wynik = parseInt(document.getElementById('scoreInput').value);

            if (!id || isNaN(wynik)) {
                alert("Wypełnij poprawne dane!");
                return;
            }

            try {
                const response = await fetch(API_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ id, wynik })
                });

                if (response.ok) {
                    alert("Wynik został zapisany!");
                    loadScores();
                } else {
                    alert("Błąd podczas zapisu wyniku.");
                }
            } catch (err) {
                console.error(err);

            }
        }

        async function loadScores() {
            try {
                // Pobieranie wyników bezpośrednio z tabeli game_scores w Supabase
                const res = await fetch(`https://vpslwxqbfemuuroyxyif.supabase.co/rest/v1/game_scores?select=*&order=score.desc&limit=5`, {
                    headers: {
                        'apikey': "sb_publishable_NWewEiCKNRdowSdOivwRvw_ywCkO1wl",
                        'Authorization': `Bearer sb_publishable_NWewEiCKNRdowSdOivwRvw_ywCkO1wl`
                    }
                });
                const data = await res.json();
                
                const list = document.getElementById('scoresList');
                list.innerHTML = "";
                
                if(data.length === 0) {
                    list.innerHTML = `<tr><td colspan="2">Brak wyników w bazie.</td></tr>`;
                    return;

                    list.innerHTML += `<tr><td>${row.client_id}</td><td><b>${row.score}</b></td></tr>`;
                });
            } catch (err) {
                console.error("Nie udało się pobrać wyników", err);
